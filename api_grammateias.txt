// GRAMMATEIA API 

const express = require('express');
const cors = require('cors');
const path = require('path');
const fs = require('fs').promises;
const mysql = require('mysql2/promise');

const app = express();
const PORT = 3000;

// middleware
app.use(cors());
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static(path.join(__dirname, 'public')));

// MySQL pool (promise)
const pool = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: '',                  // βάλε το password σου αν χρειάζεται
  database: 'diplomatikes_db',
  waitForConnections: true,
  connectionLimit: 10,
  namedPlaceholders: true
});

// helper: run query 
async function q(sql, params = {}) {
  const [rows] = await pool.query(sql, params);
  return rows;
}

// Diaxeirisi xrhstwn: kataxorisi newn foititwn / didaskon
app.post('/api/grammateia/users', async (req, res) => {
  const { onoma, epitheto, email, kodikos, rolos, am } = req.body;
  if (!onoma || !epitheto || !email || !kodikos || !rolos) {
    return res.status(400).json({ error: 'Συμπληρώστε όλα τα βασικά πεδία.' });
  }
  if (rolos === 'foititis' && !am) {
    return res.status(400).json({ error: 'Ο Αριθμός Μητρώου είναι υποχρεωτικός για φοιτητή.' });
  }
  try {
    const [result] = await q(
      `INSERT INTO xristes (onoma, epitheto, email, kodikos, rolos)
       VALUES (:onoma, :epitheto, :email, :kodikos, :rolos)`,
      { onoma, epitheto, email, kodikos, rolos }
    );
    const id = result.insertId;

    if (rolos === 'foititis') {
      await q(
        `INSERT INTO foititis (id, am) VALUES (:id, :am)`,
        { id, am }
      );
    } else if (rolos === 'didaskon') {
      await q(
        `INSERT INTO didaskon (id) VALUES (:id)`,
        { id }
      );
    }
    res.status(201).json({ message: 'Ο χρήστης καταχωρήθηκε επιτυχώς.', id });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Σφάλμα καταχώρησης χρήστη.' });
  }
});

// Provoli olwn twn diplomatikwn
app.get('/api/grammateia/diplwmatikes', async (req, res) => {
  try {
    const rows = await q(`
      SELECT d.id, d.katastasi, d.hmerominia_anathesis, d.hmerominia_exetasis,
            xf.onoma AS foititis_onoma, xf.epitheto AS foititis_epitheto, f.am,
            t.titlos
      FROM diplomatikes d
      JOIN foititis f ON d.id_foititi = f.id
      JOIN xristes xf ON f.id = xf.id
      JOIN themata t ON d.id_thematos = t.id
      ORDER BY d.id DESC
    `);
    res.json(rows);
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Σφάλμα βάσης δεδομένων' });
  }
});

// Orismos hmerominias eksetasis
app.post('/api/grammateia/orismos_exetasis', async (req, res) => {
  const { id_diplomatikis, hmerominia, wra, tropos, topothesia } = req.body;
  if (!id_diplomatikis || !hmerominia || !wra || !tropos || !topothesia) {
    return res.status(400).json({ error: 'Συμπληρώστε όλα τα πεδία.' });
  }
  try {
    await q(`
      UPDATE diplomatikes
      SET hmerominia_exetasis = :hmerominia, wra_exetasis = :wra,
          tropos_exetasis = :tropos, topothesia = :topothesia,
          katastasi = 'ypo_exetasi'
      WHERE id = :id_diplomatikis
    `, { id_diplomatikis, hmerominia, wra, tropos, topothesia });
    res.json({ message: 'Η ημερομηνία εξέτασης ορίστηκε επιτυχώς.' });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Σφάλμα βάσης δεδομένων.' });
  }
});

// AKirosi diplwmatikis + logos
app.post('/api/grammateia/akyrosi_diplwmatikis', async (req, res) => {
  const { id_diplomatikis, logos } = req.body;
  if (!id_diplomatikis || !logos) {
    return res.status(400).json({ error: 'Απαιτείται ID διπλωματικής και λόγος ακύρωσης.' });
  }
  try {
    await q(`
      UPDATE diplomatikes
      SET katastasi = 'akyromeni', logos_akyrosis = :logos
      WHERE id = :id_diplomatikis
    `, { id_diplomatikis, logos });
    res.json({ message: 'Η διπλωματική ακυρώθηκε επιτυχώς.' });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Σφάλμα βάσης δεδομένων.' });
  }
});

// Provoli praktikou eksetashs (me basi to link p dimioyrgei o didaskon)
app.get('/api/grammateia/praktiko/:id', async (req, res) => {
  const { id } = req.params;
  try {
    const [rows] = await q(
      `SELECT praktiko_link FROM diplomatikes WHERE id=:id`,
      { id }
    );
    if (!rows.length || !rows[0].praktiko_link) {
      return res.status(404).json({ error: 'Δεν βρέθηκε πρακτικό για αυτή τη διπλωματική.' });
    }
    res.json({ link: rows[0].praktiko_link });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Σφάλμα βάσης δεδομένων.' });
  }
});

// start server
app.listen(PORT, () => {
  console.log(`✅ Server running at http://localhost:${PORT}`);
});
